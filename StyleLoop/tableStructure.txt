StyleLoop ClothingStore Database Structure

Database Name: ClothingStore

Base Tables:
1. tblUser
- UserID INT AUTO_INCREMENT PRIMARY KEY
- FullName VARCHAR(100) NOT NULL
- Email VARCHAR(100) NOT NULL UNIQUE
- Username VARCHAR(50) NOT NULL UNIQUE
- PasswordHash VARCHAR(255) NOT NULL
- IsVerified TINYINT DEFAULT 0
- DateRegistered TIMESTAMP DEFAULT CURRENT_TIMESTAMP

2. tblAdmin
- AdminID INT AUTO_INCREMENT PRIMARY KEY
- AdminEmail VARCHAR(100) NOT NULL UNIQUE
- PasswordHash VARCHAR(255) NOT NULL

3. tblClothes
- ClothesID INT AUTO_INCREMENT PRIMARY KEY
- SellerID INT NULL FOREIGN KEY REFERENCES tblUser(UserID)
- ItemName VARCHAR(100) NOT NULL
- Category VARCHAR(50) NOT NULL
- Description TEXT NOT NULL
- Price DECIMAL(10,2) NOT NULL
- ConditionStatus VARCHAR(50) NOT NULL
- Size VARCHAR(30)
- ImageName VARCHAR(255) NOT NULL
- Availability VARCHAR(20) DEFAULT 'Available'

4. tblOrder
- OrderID INT AUTO_INCREMENT PRIMARY KEY
- UserID INT NULL FOREIGN KEY REFERENCES tblUser(UserID)
- ClothesID INT NULL FOREIGN KEY REFERENCES tblClothes(ClothesID)
- OrderDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP
- TotalAmount DECIMAL(10,2) NOT NULL

5. tblMessages
- MessageID INT AUTO_INCREMENT PRIMARY KEY
- SenderID INT NULL FOREIGN KEY REFERENCES tblUser(UserID)
- ReceiverID INT NULL FOREIGN KEY REFERENCES tblUser(UserID)
- MessageText TEXT NOT NULL
- DateSent TIMESTAMP DEFAULT CURRENT_TIMESTAMP
